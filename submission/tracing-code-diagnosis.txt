     1	"""FastAPI mock LLM inference service.
     2	
     3	Emits Prometheus metrics, OTLP traces, and structured JSON logs.
     4	"""
     5	from __future__ import annotations
     6	
     7	import time
     8	from contextlib import asynccontextmanager
     9	
    10	from fastapi import FastAPI, HTTPException
    11	from fastapi.responses import Response
    12	from prometheus_client import CONTENT_TYPE_LATEST, generate_latest
    13	from pydantic import BaseModel
    14	
    15	from instrumentation import (
    16	    GPU_UTIL,
    17	    INFERENCE_ACTIVE,
    18	    INFERENCE_LATENCY,
    19	    INFERENCE_QUALITY,
    20	    INFERENCE_REQUESTS,
    21	    INFERENCE_TOKENS,
    22	    bind_log,
    23	    setup_otel,
    24	    tracer,
    25	)
    26	from inference import simulate_inference, simulate_gpu_load
    27	from opentelemetry import trace
    28	
    29	
    30	@asynccontextmanager
    31	async def lifespan(_: FastAPI):
    32	    setup_otel()
    33	    yield
    34	
    35	
    36	app = FastAPI(title="day23-inference-api", lifespan=lifespan)
    37	log = bind_log("main")
    38	
    39	
    40	class PredictRequest(BaseModel):
    41	    prompt: str
    42	    model: str = "llama3-mock"
    43	    fail: bool = False  # for alert demos
    44	
    45	
    46	class PredictResponse(BaseModel):
    47	    text: str
    48	    model: str
    49	    input_tokens: int
    50	    output_tokens: int
    51	    trace_id: str
    52	    quality_score: float
    53	
    54	
    55	@app.get("/healthz")
    56	def healthz() -> dict[str, str]:
    57	    return {"status": "ok"}
    58	
    59	
    60	@app.get("/metrics")
    61	def metrics() -> Response:
    62	    GPU_UTIL.set(simulate_gpu_load())
    63	    return Response(generate_latest(), media_type=CONTENT_TYPE_LATEST)
    64	
    65	
    66	@app.post("/predict", response_model=PredictResponse)
    67	def predict(req: PredictRequest) -> PredictResponse:
    68	    INFERENCE_ACTIVE.inc()
    69	    start = time.perf_counter()
    70	    span = tracer.start_span("predict")
    71	    parent_ctx = trace.set_span_in_context(span)
    72	    span.set_attribute("gen_ai.request.model", req.model)
    73	
    74	    try:
    75	        if req.fail:
    76	            INFERENCE_REQUESTS.labels(model=req.model, status="error").inc()
    77	            log.error("forced failure", model=req.model)
    78	            raise HTTPException(status_code=503, detail="forced failure (alert demo)")
    79	
    80	        with tracer.start_as_current_span("embed-text", context=parent_ctx) as s:
    81	            s.set_attribute("text.length", len(req.prompt))
    82	            time.sleep(0.005)
    83	
    84	        with tracer.start_as_current_span("vector-search", context=parent_ctx) as s:
    85	            s.set_attribute("k", 5)
    86	            time.sleep(0.010)
    87	
    88	        with tracer.start_as_current_span("generate-tokens", context=parent_ctx) as s:
    89	            text, in_toks, out_toks, quality = simulate_inference(req.prompt, req.model)
    90	            s.set_attribute("gen_ai.usage.input_tokens", in_toks)
    91	            s.set_attribute("gen_ai.usage.output_tokens", out_toks)
    92	            s.set_attribute("gen_ai.response.finish_reason", "stop")
    93	
    94	        INFERENCE_REQUESTS.labels(model=req.model, status="ok").inc()
    95	        INFERENCE_TOKENS.labels(model=req.model, direction="input").inc(in_toks)
    96	        INFERENCE_TOKENS.labels(model=req.model, direction="output").inc(out_toks)
    97	        INFERENCE_QUALITY.labels(model=req.model).set(quality)
    98	
    99	        elapsed = time.perf_counter() - start
   100	        INFERENCE_LATENCY.labels(model=req.model).observe(elapsed)
   101	
   102	        trace_id = format(span.get_span_context().trace_id, "032x")
   103	        log.info(
   104	            "prediction served",
   105	            model=req.model,
   106	            input_tokens=in_toks,
   107	            output_tokens=out_toks,
   108	            quality=quality,
   109	            duration_seconds=round(elapsed, 4),
   110	            trace_id=trace_id,
   111	        )
   112	        return PredictResponse(
   113	            text=text,
   114	            model=req.model,
   115	            input_tokens=in_toks,
   116	            output_tokens=out_toks,
   117	            trace_id=trace_id,
   118	            quality_score=quality,
   119	        )
   120	    finally:
   121	        INFERENCE_ACTIVE.dec()
   122	        span.end()

     5	from __future__ import annotations

    19	
    20	# ── Prometheus metrics ────────────────────────────────────────
    21	INFERENCE_REQUESTS = Counter(
    22	    "inference_requests_total",
    23	    "Total inference requests",
    24	    ["model", "status"],
    25	)
    26	INFERENCE_LATENCY = Histogram(
    27	    "inference_latency_seconds",
    28	    "Inference end-to-end latency",
    29	    ["model"],
    30	    buckets=(0.05, 0.1, 0.25, 0.5, 1.0, 2.0, 5.0, 10.0),
    31	)
    32	INFERENCE_ACTIVE = Gauge(
    33	    "inference_active_gauge",
    34	    "In-flight inference requests",
    35	)
    36	INFERENCE_TOKENS = Counter(
    37	    "inference_tokens_total",
    38	    "Tokens processed (input/output)",
    39	    ["model", "direction"],
    40	)
    41	INFERENCE_QUALITY = Gauge(
    42	    "inference_quality_score",
    43	    "Latest eval-as-metric quality score [0,1]",
    44	    ["model"],
    45	)
    46	GPU_UTIL = Gauge(
    47	    "gpu_utilization_percent",
    48	    "Simulated GPU utilization [0,100]",
    49	)
    50	
    51	tracer = trace.get_tracer(__name__)
    52	
    53	
    54	def setup_otel() -> None:
    55	    """Configure OTLP trace export + FastAPI auto-instrumentation."""
    56	    resource = Resource.create(
    57	        {
    58	            "service.name": os.getenv("OTEL_SERVICE_NAME", "inference-api"),
    59	            "service.namespace": "aicb",
    60	            "deployment.environment": os.getenv(
    61	                "DEPLOY_ENV",
    62	                "lab",
    63	            ),
    64	        }
    65	    )
    66	    provider = TracerProvider(resource=resource)
    67	    endpoint = os.getenv("OTEL_EXPORTER_OTLP_ENDPOINT", "http://otel-collector:4317")
    68	    provider.add_span_processor(
    69	        BatchSpanProcessor(OTLPSpanExporter(endpoint=endpoint, insecure=True))
    70	    )
    71	    trace.set_tracer_provider(provider)
    72	    # Auto-instrument FastAPI handlers (creates server spans for every route)
    73	    from fastapi import FastAPI  # local import: only needed at setup
    74	
    75	    FastAPIInstrumentor().instrument()
    76	    _configure_logging()
    77	
    78	
    79	def _configure_logging() -> None:
    80	    logging.basicConfig(
    81	        format="%(message)s",
    82	        stream=sys.stdout,
    83	        level=os.getenv("LOG_LEVEL", "INFO"),
    84	    )
    85	    structlog.configure(
    86	        processors=[
    87	            structlog.contextvars.merge_contextvars,
    88	            structlog.processors.add_log_level,
    89	            structlog.processors.TimeStamper(fmt="iso"),
    90	            structlog.processors.JSONRenderer(),
    91	        ],
    92	        wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
    93	        logger_factory=structlog.PrintLoggerFactory(),
    94	        cache_logger_on_first_use=True,
    95	    )
    96	
    97	
    98	def bind_log(name: str) -> structlog.BoundLogger:
    99	    return structlog.get_logger(name)
